* External Services

Packages that connect Emacs to external services (Jira, GitLab, etc.). Each
uses =use-package :ensure t= so installation and configuration are co-located.
Auth credentials live in =~/.authinfo.gpg= — no plaintext tokens in config.

** Jira — org-jira

Syncs Jira tickets into org-mode files via =org-jira-get-issues-from-custom-jql=.
Tickets land in =org-jira-working-dir= (=~/Code/Org/=).

#+begin_src emacs-lisp :tangle customizations/external-services.el
;;; external-services.el --- Customizations -*- lexical-binding: t; -*-
;; org 9.1.x (org-20171225) does not define org-show-all; shim for org-jira compatibility.
;; Always define — no guard — so hot-reloads pick up the correct target (outline-show-all).
(defalias 'org-show-all 'outline-show-all)

(use-package org-jira
  :ensure t
  :config
  (setq jiralib-url "https://rentable.atlassian.net"
        jiralib-user-login-name "matthew@joinroost.com"
        org-jira-working-dir "~/Code/Org/"
        org-jira-use-status-as-todo t
        org-jira-project-filename-alist '(("SUP-backlog" . "JIRA.org"))
        org-jira-custom-jqls
        '((:jql "project in (ROOST, SUP) AND assignee = currentUser() AND status not in (Closed, Done, Backlog) ORDER BY priority DESC"
                :limit 100 :filename "JIRA")
          (:jql "project = SUP AND status != Done AND priority != Lowest AND (assignee IS EMPTY OR assignee NOT IN (\"5db85c6a4ff1890d3103526d\", \"60c7c5e4dae567006837e666\", \"712020:574d1349-793b-4adc-94b6-85016462b3f9\", \"712020:5d190e7c-b74b-404d-b310-b9f6cf581f3c\")) ORDER BY priority DESC, created DESC"
                :limit 100 :filename "SUP-backlog")))

  ;; Color-code Jira statuses so they're visually distinct
  (setq org-todo-keyword-faces
        '(("TODO"            . (:foreground "#00B000" :weight bold))
          ("BLOCKED"         . (:foreground "#ff2222" :weight bold))
          ("BACKLOG"         . (:foreground "#888888" :weight normal))
          ("DESIGN/SPEC"     . (:foreground "#c792ea" :weight bold))
          ("IN-PROGRESS"     . (:foreground "#228B22" :weight bold))
          ("DESK-REVIEW"     . (:foreground "#ffcb6b" :weight bold))
          ("CODE-REVIEW"     . (:foreground "#82aaff" :weight bold))
          ("IN-REVIEW"       . (:foreground "#89ddff" :weight bold))
          ("READY-TO-DEPLOY" . (:foreground "#c3e88d" :weight bold))
          ("DONE"            . (:foreground "#555555" :weight normal))
          ("CLOSED"          . (:foreground "#555555" :weight normal))
          ;; GitLab MR review states (bebop-gitlab.el)
          ("OPEN"            . (:foreground "#ff9e64" :weight bold))
          ("RESOLVED"        . (:foreground "#555555" :weight normal))))

  ;; GitLab MR review states must be registered as TODO keywords so that
  ;; org-todo, org-get-todo-state, and heading fontification all work.
  (add-to-list 'org-todo-keywords '(sequence "OPEN" "|" "RESOLVED") t)

  ;; org-jira switches to the project buffer at the end of every render.
  ;; Wrap it so the window configuration is restored when the callback completes,
  ;; and save the buffer so tickets persist across close/reopen.
  (advice-add 'org-jira--render-issues-from-issue-list :around
    (lambda (orig-fn issues)
      (save-window-excursion (funcall orig-fn issues))
      (with-current-buffer (org-jira--get-project-buffer (-last-item issues))
        (save-buffer))))

  ;; Override the per-project heading logic.  Route SUP-backlog issues
  ;; under "* SUPs"; everything else under "* Tickets".
  (advice-add 'org-jira--maybe-render-top-heading :override
    (lambda (_proj-key)
      (let ((heading (if (equal org-jira-proj-key-override "SUP-backlog")
                         "Engineering SUPs" "My Tickets")))
        (goto-char (point-min))
        (unless (re-search-forward (format "^\\* %s" heading) nil t)
          (goto-char (point-max))
          (insert (format "\n\n* %s\n" heading)))
        (goto-char (point-min))
        (re-search-forward (format "^\\* %s" heading) nil t))))

  ;; Prepend the Jira key to the heading, padded to a fixed width so
  ;; descriptions align across tickets.  Named function prevents stacking
  ;; when the config is reloaded.
  (defun jira--prepend-slug (issue)
    "Prepend a fixed-width issue key to ISSUE headline.
Left-pad compensates for TODO keyword width so slugs align;
slug is padded to a fixed width so descriptions align."
    (let* ((id       (slot-value issue 'issue-id))
           (hl       (slot-value issue 'headline))
           (status   (slot-value issue 'status))
           (kw-pad   (make-string (max 0 (- 15 (length status))) ?\s))
           (slug-col (format "%-12s" id)))
      (unless (string-match-p (regexp-quote id) hl)
        (setf (slot-value issue 'headline)
              (format "%s%s  %s" kw-pad slug-col hl)))))
  (advice-add 'org-jira--render-issue :before #'jira--prepend-slug)

  (defalias 'jira-progress #'org-jira-progress-issue)

  (defun jira--clear-section (heading)
    "Clear the section under HEADING in JIRA.org, if present."
    (let ((buf (find-file-noselect (expand-file-name "JIRA.org" org-jira-working-dir))))
      (with-current-buffer buf
        (save-excursion
          (goto-char (point-min))
          (when (re-search-forward (format "^\\* %s" heading) nil t)
            (let ((start (line-beginning-position)))
              (org-end-of-subtree t t)
              (delete-region start (point)))))
        (save-buffer))))

  (defun jira-sync ()
    "Clear Tickets and SUPs sections in JIRA.org, then fetch fresh from Jira.
Each query is fired explicitly — org-jira's async chaining doesn't
reliably process multiple custom-jql entries."
    (interactive)
    (jira--clear-section "My Tickets")
    (jira--clear-section "Engineering SUPs")
    ;; jiralib--agile-call-async emits several progress messages on every page
    ;; of results.  Suppress all messages for the sync window, then restore.
    (setq inhibit-message t)
    ;; Fire each JQL entry separately with staggered timers.
    (let ((tickets (list (car org-jira-custom-jqls)))
          (sups    (cdr org-jira-custom-jqls)))
      (org-jira-get-issues-from-custom-jql tickets)
      (run-with-timer 5 nil #'org-jira-get-issues-from-custom-jql tickets)
      (run-with-timer 8 nil #'org-jira-get-issues-from-custom-jql sups)
      (run-with-timer 13 nil #'org-jira-get-issues-from-custom-jql sups)
      (run-with-timer 18 nil (lambda () (setq inhibit-message nil)))))
  (defun jira--sort-section (heading)
    "Sort HEADING section by TODO status, then collapse subtrees."
    (goto-char (point-min))
    (when (re-search-forward (format "^\\* %s" heading) nil t)
      (org-sort-entries nil ?O)
      (goto-char (point-min))
      (re-search-forward (format "^\\* %s" heading) nil t)
      (outline-hide-subtree)
      (outline-show-children)))

  (defun jira-sort ()
    "Sort Tickets and SUPs sections by TODO status.
Re-collapses ticket subtrees after sorting so descriptions stay hidden."
    (interactive)
    (with-current-buffer (find-file-noselect
                          (expand-file-name "JIRA.org" org-jira-working-dir))
      (save-excursion
        (jira--sort-section "My Tickets")
        (jira--sort-section "Engineering SUPs"))))

  (with-eval-after-load 'org
    (define-key org-mode-map (kbd "C-c S") #'jira-sort)))
#+end_src

Use =M-x jira-sync= to fetch tickets. =C-c s= sorts the =* Tickets= section
with DONE/CLOSED at the top and TODO at the bottom.

** GitLab — forge

Magit extension with native GitLab (and GitHub) support. Surfaces MRs, issues,
and review comments as Magit sections; supports commenting, approving, and
merging from within Emacs.

Add an entry to =~/.authinfo.gpg= (the =^forge= suffix is required):

: machine gitlab.com login YOUR_USERNAME^forge password YOUR_PERSONAL_ACCESS_TOKEN

Token needs at minimum =api= scope.

#+begin_src emacs-lisp :tangle customizations/external-services.el
(use-package forge
  :ensure t
  :after magit)
#+end_src

Verify with =M-x forge-pull= in a Magit buffer for a GitLab repo.
